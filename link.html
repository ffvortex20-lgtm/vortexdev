// Lógica do Redirecionador (link.html) async function setupRedirect() { const params = new URLSearchParams(window.location.search); const targetUrl = params.get('url'); // Link original const productName = params.get('name'); document.getElementById('title').innerText = productName; // Simulação de "Unlock" (Estilo Rekonise) const btn = document.getElementById('unlock-btn'); btn.onclick = async () => { btn.innerText = "Checking..."; // Aqui entra a API do TinyURL para gerar o link final curto const response = await fetch(`https://tinyurl.com/api-create.php?url=${encodeURIComponent(targetUrl)}`); const shortLink = await response.text(); setTimeout(() => { window.location.href = shortLink; }, 3000); // 3 segundos de "validação" }; }
